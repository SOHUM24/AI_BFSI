// ===== UTILITIES =====
const fmt = n => n == null || isNaN(n) ? '—' : n.toLocaleString('en-IN', {maximumFractionDigits:0});
const fmt1 = n => n == null || isNaN(n) ? '—' : n.toLocaleString('en-IN', {maximumFractionDigits:1});
const fmt2 = n => n == null || isNaN(n) ? '—' : n.toLocaleString('en-IN', {maximumFractionDigits:2});
const cr = n => n == null ? null : n / 100; // lakhs to crores

const STATES = ['ANDHRA PRADESH','BIHAR','CHHATTISGARH','GUJARAT','HARYANA','HIMACHAL PRADESH','JAMMU AND KASHMIR','JHARKHAND','KARNATAKA','KERALA','MADHYA PRADESH','MAHARASHTRA','ORISSA','PUNJAB','RAJASTHAN','TAMIL NADU','TELANGANA','UTTAR PRADESH','UTTARAKHAND','WEST BENGAL'];

// ===== COLOR SCALE FOR MAP =====
function getColor(value) {
  if (value == null) return '#ccc';
  const crv = value / 100; // in crores
  if (crv > 500000) return '#0b2318';
  if (crv > 200000) return '#1f5438';
  if (crv > 50000)  return '#3a8c5d';
  return '#7bc49a';
}

// ===== NATIONAL STATS =====
function setNationalStats() {
  const ai = DCCB_DATA['ALL_INDIA'];
  if (!ai) return;
  document.getElementById('nat-dccbs').textContent = fmt(ai.dccbs);
  document.getElementById('nat-deposits').textContent = fmt2(cr(ai.deposits));
  document.getElementById('nat-loans').textContent = fmt2(cr(ai.loans));
  document.getElementById('nat-business').textContent = fmt2(cr(ai.business));
  document.getElementById('nat-members').textContent = fmt(ai.members);
  document.getElementById('nat-employees').textContent = fmt(ai.employees);

  document.getElementById('ai-dccbs').textContent = fmt(ai.dccbs);
  document.getElementById('ai-offices').textContent = fmt(ai.offices);
  document.getElementById('ai-members').textContent = fmt(ai.members);
  document.getElementById('ai-employees').textContent = fmt(ai.employees);
  document.getElementById('ai-deposits').textContent = fmt2(ai.derived.deposits_in_cr);
  document.getElementById('ai-loans').textContent = fmt2(ai.derived.loans_in_cr);
  document.getElementById('ai-cdr').textContent = fmt1(ai.derived.cd_ratio);
  const plEl = document.getElementById('ai-pl');
  plEl.textContent = (ai.pl >= 0 ? '+' : '') + fmt2(ai.derived.loans_in_cr > 0 ? ai.pl / 100 : ai.pl);
  plEl.className = 'panel-stat-val ' + (ai.pl >= 0 ? '' : 'negative');
}

// ===== MAP COLORING =====
function colorMap() {
  STATES.forEach(state => {
    const d = DCCB_DATA[state];
    const el = document.querySelector(`.map-state[data-state="${state}"]`);
    if (el && d && d.business != null) {
      el.style.fill = getColor(d.business);
    }
  });
}

// ===== MINI STATS =====
function setMiniStats() {
  let topDep = {s:'',v:0}, topLoan = {s:'',v:0}, topDccbs = {s:'',v:0}, topMem = {s:'',v:0};
  STATES.forEach(s => {
    const d = DCCB_DATA[s];
    if (!d) return;
    if (d.deposits > topDep.v) topDep = {s, v:d.deposits};
    if (d.loans > topLoan.v) topLoan = {s, v:d.loans};
    if (d.dccbs > topDccbs.v) topDccbs = {s, v:d.dccbs};
    if (d.members > topMem.v) topMem = {s, v:d.members};
  });
  document.getElementById('top-state-deposits').textContent = topDep.s.split(' ')[0];
  document.getElementById('top-state-loans').textContent = topLoan.s.split(' ')[0];
  document.getElementById('top-state-dccbs').textContent = topDccbs.s.split(' ')[0];
  document.getElementById('top-state-members').textContent = topMem.s.split(' ')[0];
}

// ===== DATA TABLE =====
function buildTable() {
  const tbody = document.getElementById('table-body');
  let html = '';
  STATES.forEach(state => {
    const d = DCCB_DATA[state];
    if (!d) return;
    const npaPct = d.derived.npa_to_loans_pct;
    const plClass = d.pl >= 0 ? 'pos' : 'neg';
    const kccText = d.kcc_sanctioned != null ? `₹${fmt(d.kcc_sanctioned)} L` : '—';
    html += `<tr>
      <td>${state}</td>
      <td class="num">${fmt(d.dccbs)}</td>
      <td class="num">${fmt(d.offices)}</td>
      <td class="num">${fmt(d.employees)}</td>
      <td class="num">${fmt(d.members)}</td>
      <td class="num">${fmt2(cr(d.deposits))}</td>
      <td class="num">${fmt2(cr(d.loans))}</td>
      <td class="num">${fmt2(cr(d.business))}</td>
      <td class="num">${fmt1(d.derived.cd_ratio)}%</td>
      <td class="num">${fmt2(cr(d.npa))}</td>
      <td class="num ${npaPct > 15 ? 'neg' : (npaPct < 5 ? 'pos' : '')}">${fmt1(npaPct)}%</td>
      <td class="num ${plClass}">${d.pl >= 0 ? '+' : ''}${fmt2(cr(d.pl))}</td>
      <td class="num">${kccText}</td>
    </tr>`;
  });
  tbody.innerHTML = html;
}
